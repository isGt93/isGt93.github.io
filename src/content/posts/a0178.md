if contains 'user' ${!URL_PARAMS[@]} && file_exists "users/${URL_PARAMS['user']}"; then



    # 读取用户文件的第一行作为用户名

    local username=$(head -n 1 "users/${URL_PARAMS['user']}");



    # 输出 HTML 标题，显示用户名的帖子

    echo "<h3>${username}'s posts:</h3>";



    # 开始一个有序列表

    echo "<ol>";



    # 获取并遍历该用户的帖子列表

    get_users_posts "${username}" | while read -r post; do

        # 使用 awk 处理帖子路径，生成链接 slug

        post_slug=$(awk -F/ '{print $2 "#" $3}' <<< "${post}");



        # 输出每个帖子为列表项，其中包含指向帖子详情页的链接

        echo "<li><a href=\"/post.wtf?post=${post_slug}\">$(nth_line 2 "${post}" | htmlentities)</a></li>";

    done 



    # 结束有序列表

    echo "</ol>";



    # 如果当前用户已登录、是管理员并且查看的是自己的资料，则调用 get_flag1 函数

    if is_logged_in && [[ "${COOKIES['USERNAME']}" == 'admin' ]] && [[ "${username}" == 'admin' ]]; then

        get_flag1;

    fi

fi

include\_page 函数告诉我们，`.wtf`文件是可以 eval 执行代码的， 只需要行首是`$` 符号

```
function include_page {

    # include_page <pathname>

    local pathname=$1  # 获取传入的第一个参数作为路径名

    local cmd=""       # 初始化一个空字符串用于累积脚本命令行

    local can_execute=1  # 假设文件默认不可执行



    # 检查文件是否以 .wtf 结尾，如果是则设置 can_execute 为 0（表示可执行）

    if [[ "${pathname:(-4)}" == '.wtf' ]]; then

        can_execute=0

    fi



    page_include_depth=$(($page_include_depth+1))  # 增加包含深度计数器

    if [[ $page_include_depth -lt $max_page_include_depth ]]; then

        local line;

        while read -r line; do

            # 检查行是否以 $ 符号开头（意味着是脚本行），并且文件是可以执行的

            local is_script=1

            if [[ "$" == "${line:0:1}" && ${can_execute} == 0 ]]; then

                is_script=0

            fi



            # 如果是脚本行，则累加到 cmd 变量中

            if [[ $is_script == 0 ]]; then

                cmd+=;

'"${line#"$"}";  # 移除行首的 $ 并添加到 cmd 中

            else

                # 如果之前有累积的命令，则尝试执行它们，并在发生错误时记录日志

                if [[ -n $cmd ]]; then

                    eval "$cmd" || echo "Error during execution of ${cmd}";  # 注意：命令执行

                    cmd=""

                fi

                # 输出非脚本行的内容

                echo "$line"

            fi

        done < ${pathname}  # 从指定路径名的文件中读取内容

    else

        # 如果达到了最大嵌套深度，输出错误信息

        echo "<p>Max include depth exceeded!</p>"

    fi

}
```

reply 函数告诉我们，回复帖子会创建一个新的文件，路径是`posts/${post_id}/${next_reply_id}` ，对应接口是 `/reply.wtf?post=${post_id}` ，使用奇怪的`%09`截断，就能得到具体的文件路径，文件内容是用户名和帖子追加的回复内容，这些内容都是可以用户控制的（亲测，写到username中的命令可以看见执行结果，写到text中的命令看不见执行结果，不知道为啥）

```
function reply {

    # 回复函数接收三个参数：

    # $1 - post_id: 帖子的唯一标识符

    # $2 - username: 发布回复的用户的用户名

    # $3 - text: 回复的内容



    local post_id=$1;       # 将第一个参数赋值给局部变量 post_id

    local username=$2;      # 将第二个参数赋值给局部变量 username

    local text=$3;          # 将第三个参数赋值给局部变量 text

    local hashed=$(hash_username "${username}");  # 调用 hash_username 函数对用户名进行哈希处理，并将结果存入局部变量 hashed



    # 获取当前帖子目录下的最大文件名（即最后一个回复ID），通过遍历 posts/${post_id} 目录下所有文件，获取其基础名称并排序后取最后一条

    curr_id=$(for d in posts/${post_id}/*; do basename "$d"; done | sort -n | tail -n 1);



    # 计算下一个回复的 ID，基于当前最大的 ID 加一

    next_reply_id=$(awk '{print $1+1}' <<< "${curr_id}");



    # 构建下一个回复文件的完整路径

    next_file="posts/${post_id}/${next_reply_id}"; # 注意：奇怪的%09截断待会作用于此



    # 向新的回复文件中写入用户名

    echo "${username}" > "${next_file}";



    # 向新的回复文件中追加原始帖子的主题（假设第二行是主题）

    echo "RE: $(nth_line 2 < "posts/${post_id}/1")" >> "${next_file}";



    # 向新的回复文件中追加回复内容

    echo "${text}" >> "${next_file}";



    # 将新创建的回复添加到用户查找表（users_lookup）中对应用户的帖子列表里

    # 这样可以通过用户的哈希值找到他们所参与的所有帖子

    echo "${post_id}/${next_reply_id}" >> "users_lookup/${hashed}/posts";

}
```
