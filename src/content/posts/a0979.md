ailx10说：这篇是在2018年11月之前写的，后来又改过。单独分 XSS 已经试过几轮，我开始想把正常、XSS、SQL 和 PHP 都塞进同一个分发模型，让它先判断报文大概往哪类引擎送。TFLearn 的 LSTM 训练、保存和预测都接上后，前几类能分，PHP 却没认好，单条速度也不算理想。我把这些表现一起记下来。它是从算法练习往 WAF 结构走的一步：不再只问一条字符串像不像攻击，而是开始安排一个模型在整个流程里究竟负责哪件事。

---

技术栈：`TFlearn + RNN + LSTM`

训练样本：正常报文：135k，XSS报文：16k，SQL报文：10k，PHP报文：1.5k。

模型准确率：96%，能识别正常报文，XSS报文，SQL报文，识别PHP报文失败。

模型性能：单一样本10ms一条，建议批量处理样本提高识别速度。

模型持久化：一次训练，到处使用。

测试数据：


完整代码：

```
import sys

import urllib

import numpy as np

import tensorflow as tf

import tflearn

from tflearn.data_utils import to_categorical, pad_sequences

from sklearn.model_selection import train_test_split

import time

def elt(line):

    x = []

    for i, c in enumerate(line):

        c = c.lower()

        

        x.append(ord(c))

    return x



def load_file(filename,label,ms=[],ns=[]):

    with open(filename) as f:

        for line in f:

            line = line.strip('

')

            line = urllib.unquote(line)

            m = elt(line)

            if(label == 3):

                n = 3

            elif(label == 2):

                n = 2

            elif(label == 1):

                n = 1

            else:

                n = 0

            ms.append(m)

            ns.append(n)

        print(len(ns))



def load_files(file1,file2,file3,file4):

    xs = []

    ys = []

    load_file(file1,0,xs,ys)

    load_file(file2,1,xs,ys)

    load_file(file3,2,xs,ys)

    load_file(file4,3,xs,ys)

    return xs,ys



def one_hot(x,size):

    v = []

    for x1 in x:

        x2 = [0]*size

        x2[x1] = 1

        v.append(x2)

    return v



def train(x,y):

   graph1 = tf.Graph()

   with graph1.as_default():

        x_train, x_test, y_train, y_test=train_test_split( x,y, test_size=0.4,random_state=0)



    	x_train = pad_sequences(x_train,maxlen=100,value=0.)

    	x_test = pad_sequences(x_test,maxlen=100,value=0.)

#        y_train = one_hot(y_train,size=3)

#        y_test = one_hot(y_test,size=3)

        y_train = to_categorical(y_train, nb_classes=4)

        y_test = to_categorical(y_test, nb_classes=4)



        net = tflearn.input_data([None, 100])

        net = tflearn.embedding(net, input_dim=256, output_dim=128)

        net = tflearn.lstm(net, 128, dropout=0.8)

        net = tflearn.fully_connected(net, 4, activation='softmax')

        net = tflearn.regression(net, optimizer='adam', learning_rate=0.1,

                                 loss='categorical_crossentropy')

        model = tflearn.DNN(net, tensorboard_verbose=3)

        model.fit(x_train, y_train,n_epoch=1, validation_set=(x_test, y_test), show_metric=True,

                 batch_size=200,run_id="gutao")

        model.save("./gut_php.tfl")



def test():

   graph2 = tf.Graph()

   with graph2.as_default():

        net = tflearn.input_data([None, 100])

        net = tflearn.embedding(net, input_dim=256, output_dim=128)

        net = tflearn.lstm(net, 128, dropout=0.8)

        net = tflearn.fully_connected(net, 4, activation='softmax')

        net = tflearn.regression(net, optimizer='adam', learning_rate=0.1,

                                 loss='categorical_crossentropy')

        model = tflearn.DNN(net, tensorboard_verbose=3)

        model.load("./gut_php.tfl")

        normal = "/using-localization/"

        x_xss = "/0_1/include/dialog/config.php?adminDirHand=%22/%3E%3C/script%3E%3Cscript%3Ealert(1);%3C/script%3E"

        x_xss = "<script>alert(1)</script>"

        x_sql = "/wp-login.php?action=lostpassword%25%27%20ORDER%20BY%201%23"

        x_sql = "' and 1=1 #"

        x_php = "ex1.php?dir=| cat /etc/passwd"

        normal = elt(urllib.unquote(normal))

        x_xss = elt(urllib.unquote(x_xss))

        x_sql = elt(urllib.unquote(x_sql))

        x_php = elt(urllib.unquote(x_php))

    	normal = pad_sequences([normal],maxlen=100,value=0.)

    	x_xss = pad_sequences([x_xss],maxlen=100,value=0.)

    	x_sql = pad_sequences([x_sql],maxlen=100,value=0.)

    	x_php = pad_sequences([x_php],maxlen=100,value=0.)

        o1 = o2 = o3 = o4 = o5 = 0

        s1 = s2 = s3 = s4 = 0

        x = 0

        while(x < 1000):

            o1 = time.clock()

            p1 = model.predict_label(normal)

            o2 = time.clock()

            p2 = model.predict_label(x_xss)

            o3 = time.clock()

            p3 = model.predict_label(x_sql)

            o4 = time.clock()

            p4 = model.predict_label(x_php)

            o5 = time.clock()

            s1 += o2 - o1

            s2 += o3 - o2

            s3 += o4 - o3

            s4 += o5 - o4

            x += 1

        print(s1,s2,s3,s4)

        print(p1,p2,p3,p4)



if __name__ == "__main__":

    xs,ys = load_files(sys.argv[1],sys.argv[2],sys.argv[3],sys.argv[4])

    train(xs,ys)

    test()
```

PHP识别没有达到预期，是这篇最该留下的一段。把每类表现分别看，模型才不会因为总体分数不错，就把一个始终没学好的方向悄悄藏起来。
