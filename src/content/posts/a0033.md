## Epoll:

```
/********************************/
#include <sys/epoll.h>
typedef void (*EP_CALLBACK_FUNC)(int event,int fd);
typedef struct tagEpEntry
{
  int fd;
  EP_CALLBACK_FUNC fCallBack;
} EP_ENTRY_S;

void ep_init();
void ep_add(EP_ENTRY_S* entry);
void ep_del(EP_ENTRY_S* entry);
void ep_exit();
void ep_wait();

/********************************/

#include <stdio.h>
#include "epoll.h"

static int gEpollFd = 0;

void ep_init()
{
  int epfd = epoll_create(1);
  gEpollFd = epfd;
}

void ep_add(EP_ENTRY_S* entry)
{
  struct epoll_event stEpollEvent;
  stEpollEvent.events = EPOLLIN | EPOLLHUP | EPOLLERR;
  stEpollEvent.data.ptr = (void*)entry;
  epoll_ctl(gEpollFd,EPOLL_CTL_ADD,entry->fd,&stEpollEvent);
}

void ep_del(EP_ENTRY_S* entry)
{
  struct epoll_event stEpollEvent;
  stEpollEvent.events = EPOLLIN | EPOLLHUP | EPOLLERR;
  stEpollEvent.data.ptr = (void*)entry;
  epoll_ctl(gEpollFd,EPOLL_CTL_DEL,entry->fd,&stEpollEvent);
}

void ep_exit()
{
  gEpollFd = -1;
}

void ep_wait()
{
  int fd;
  int num;
  EP_ENTRY_S* entry;
  EP_CALLBACK_FUNC fCallBack;
  struct epoll_event astEpollEvent[1];
  while(1)
  {
    fd = epoll_wait(gEpollFd,astEpollEvent,1,-1);
    for(num = 0;num < fd;++num)
    {
      entry = (EP_ENTRY_S*)astEpollEvent[num].data.ptr;
      fCallBack = entry->fCallBack;
      fCallBack(astEpollEvent[num].events,entry->fd);
    }
  }
}
```

## Timer:

```
/********************************/
#include <stdio.h>
#include <sys/timerfd.h>

FILE* tick_start();
void tick_happy(FILE*);
void tick_end(FILE*);
void timer_init();
void timer_over();

/********************************/
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include "epoll.h"
#include "timer.h"

static EP_ENTRY_S gEpollEntry = {-1,NULL};
static int gTickNum = 0;
static char gBuff[16];

FILE* tick_start()
{
  FILE* happyfd = fopen("happy","r");
  return happyfd;
}

void tick_happy(FILE* happyfd)
{
  char buff[16];
  fread(buff,sizeof(char),40,happyfd);
  memcpy(gBuff,buff,sizeof(gBuff));
}

void tick_end(FILE* happyfd)
{
  fclose(happyfd);
}

static void _callback(int event,int fd)
{
  uint64_t count = 0;
  if(0 != (EPOLLIN & event))
  {
    if(0 < read(fd,(void*)&count,sizeof(count)))
    {
      ++gTickNum;
      if(gTickNum < 12)
        printf("%d tick
",gTickNum);
      else
      {
        printf("%d tick
%s
",gTickNum,gBuff);
        exit(0);
      }
    }
  }
}

void timer_init()
{
  struct itimerspec timer;
  memset(&timer,0,sizeof(timer));
  int timerfd = timerfd_create(CLOCK_MONOTONIC,0);
  timer.it_interval.tv_sec = 1;
  timer.it_value.tv_sec = 1;
  timerfd_settime(timerfd,0,&timer,NULL);
  gEpollEntry.fd = timerfd;
  gEpollEntry.fCallBack = _callback;
  ep_add(&gEpollEntry);
}

void timer_over()
{
  ep_del(&gEpollEntry);
  close(gEpollEntry.fd);
  gEpollEntry.fCallBack = NULL;
}
```
