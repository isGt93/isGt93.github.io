static ngx_int_t ngx_http_aisecurity_init(ngx_conf_t *cf)

{

    ngx_http_handler_pt* h_rewrite;

    ngx_http_core_main_conf_t* cmcf;

    

    cmcf = ngx_http_conf_get_module_main_conf(cf,ngx_http_core_module);

    if(cmcf == NULL) return NGX_ERROR;

    h_rewrite = ngx_array_push(&cmcf->phases[NGX_HTTP_REWRITE_PHASE].handlers);

    if(h_rewrite == NULL) return NGX_ERROR;

    *h_rewrite = ngx_http_aisecurity_rewrite_handler;

    

    return NGX_OK;

}



static void* ngx_http_aisecurity_create_main_conf(ngx_conf_t* cf)

{

    ngx_http_aisecurity_conf_t* conf;

    conf = ngx_http_aisecurity_create_conf(cf);

    if(conf == NULL || conf == NGX_CONF_ERROR) return NGX_CONF_ERROR;

    

    PyObject* pModule = aisec_waf_init();

    PyObject* pEngine = aisec_waf_load(pModule);

    conf->pModule = pModule;

    conf->pEngine = pEngine;

    printf("[engine info] pModule=%p	pEngine=%p

",conf->pModule,conf->pEngine);

    if(conf->pEngine == NULL) return NGX_CONF_ERROR;

    return conf;

}



static char* ngx_http_aisecurity_init_main_conf(ngx_conf_t* cf,void* conf)

{

    return NGX_CONF_OK;

}



static void* ngx_http_aisecurity_create_conf(ngx_conf_t* cf)

{

    ngx_http_aisecurity_conf_t* conf;

    conf = (ngx_http_aisecurity_conf_t*)ngx_palloc(cf->pool,sizeof(ngx_http_aisecurity_conf_t));

    if(conf == NULL) return NGX_CONF_ERROR;

    conf->enable = NGX_CONF_UNSET;

    conf->pool = cf->pool;

    return conf;

}



static char* ngx_http_aisecurity_merge_srv_conf(ngx_conf_t* cf,void* parent,void* child)

{

    ngx_http_aisecurity_conf_t* p = parent;

    ngx_http_aisecurity_conf_t* c = child;

    ngx_conf_merge_value(c->enable,p->enable,0);

    return NGX_CONF_OK;

}



static char* ngx_http_aisecurity_merge_loc_conf(ngx_conf_t* cf,void* parent,void* child)

{

    ngx_http_aisecurity_conf_t* p = parent;

    ngx_http_aisecurity_conf_t* c = child;

    ngx_conf_merge_value(c->enable,p->enable,0);

    return NGX_CONF_OK;

}



ngx_int_t ngx_http_aisecurity_rewrite_handler(ngx_http_request_t *r)

{

    //return NGX_DECLINED;

    printf("rewrite_handler

");

    //return NGX_DECLINED;

    ngx_int_t atk = 0;

    ngx_int_t ret = 0;

    ngx_http_aisecurity_conf_t* mlcf;

    ngx_http_aisecurity_conf_t* mmcf;

    mlcf = ngx_http_get_module_loc_conf(r, ngx_http_aisecurity_module);

    mmcf = ngx_http_get_module_main_conf(r, ngx_http_aisecurity_module);

    printf("cf:%d	%p	%p

",(int)mlcf->enable,mmcf->pModule,mmcf->pEngine);

    if(mlcf->enable != 1) return NGX_DECLINED;

    const char* uri = ngx_str_to_char(r->unparsed_uri,r->pool);

    const char* method = ngx_str_to_char(r->method_name,r->pool);

    const char** pkt_addr = get_data(uri,method);

    

    atk = aisec_process_uri(mmcf,pkt_addr);

    free_data(pkt_addr);

    if(atk) ret = NGX_HTTP_FORBIDDEN;

    if(ret) return ret;

    

    return NGX_DECLINED;

}



const char** get_data(const char* uri,const char* method)

{

    char** pkt_addr = (char**)malloc(2*sizeof(char*));

    pkt_addr[0] = (char*)uri;

    pkt_addr[1] = (char*)method;

    printf("pkt:

%s

%s

",pkt_addr[0],pkt_addr[1]);

    return (const char**)pkt_addr;

}



void free_data(const char** pkt_addr)

{

    free(pkt_addr);

    return;

}



ngx_int_t aisec_process_uri(ngx_http_aisecurity_conf_t* cf,const char** pkt_loads)

{

    ngx_int_t ret = 0;

    ret = aisec_waf_predict(cf->pModule,cf->pEngine,pkt_loads);

    return ret;

}



ngx_inline char *ngx_str_to_char(ngx_str_t a, ngx_pool_t *p)

{

    char *str = NULL;

    if (a.len == 0) {

        return NULL;

    }

    str = ngx_pnalloc(p, a.len+1);

    if (str == NULL) {

        return (char *)-1;

    }

    ngx_memcpy(str, a.data, a.len);

    str[a.len] = '\0';

    return str;

}
