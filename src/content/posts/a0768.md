ailx10说：这篇是在2024年8月之前写的，后来又改过。为了做安全演示，不想每次都真跑矿工，我就自己写了两段 Python，分别扮演客户端和矿池。登录、下发任务、尝试 nonce、提交再确认，流程接起来后，还能用 Wireshark 看见一段像样的通信，CPU 也会在模拟计算时忙起来。这篇就记了这套小道具怎样搭出来。它好玩的地方是把原本只在告警里看到的“矿池连接”，变成自己能控制速度和难度的过程，讲解时终于可以停在任何一步，看每条消息为什么会出现。

---

本文记录一个 Python 本地客户端与服务端的类矿池通信模拟。它用于理解消息交互，并非真实 XMRig 或 RandomX 的完整实现。

def create_socket_connection(host='127.0.0.1', port=3333):
    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

    sock.listen(1)
    print("矿池启动...")
    conn, addr = sock.accept()
    print('矿工上线:{}'.format(addr))
    return conn

    sock.sendall(json.dumps(data).encode() + b'

    sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
    sock.connect((host, port))
    return sock

def send_json(sock, data):
    sock.sendall(json.dumps(data).encode() + b'
')

def receive_json(sock):
    data = b''
    while not data.endswith(b'
'):
        data += sock.recv(1)
    return json.loads(data.decode())

def mine():
    blob = "1010f4d5d0b5066cb8820463497fcfc3cf7eb6a5cca34248de2b0c4c035c332cfdcaed48e6602200000000246f9168ec42820a6fa0d48b440e7491e70413efd54abed6f1d6daae9790a75218"
    target = "ffffffffffffffffffffffffffffffffffffffffffffffffffffff0000"

    blob_bytes = bytes.fromhex(blob)
    target_bytes = bytes.fromhex(target)

    # 解析目标值为目标难度
    target_difficulty = int.from_bytes(target_bytes, byteorder='big')
    nonce = 0
    while True:
        # 拼接 blob 和 nonce
        data_to_hash = blob_bytes + struct.pack('<I', nonce)  # 使用小端格式将 nonce 添加到 blob 后面
        # time.sleep(1)
        # 假设的哈希函数（此处使用 sha256 代替 CryptoNight）
        hash_result = hashlib.sha256(data_to_hash).digest()
        int_result = int.from_bytes(hash_result, byteorder='big')
        # 检查是否满足目标难度
        if int_result < target_difficulty:
            print(f"Found hash: {hash_result.hex()} with nonce: {nonce}")
            return hash_result.hex(), nonce
        # 如果未达到难度，则增加 nonce 并继续尝试
        nonce += 1

def main():
