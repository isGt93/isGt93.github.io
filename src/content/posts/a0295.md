```
└─# echo -n 'ailx10'|md5sum|cut -d ' ' -f1

83b70504e0d8742dd5b66e6962eb8a35



update tblUsers set pwd="83b70504e0d8742dd5b66e6962eb8a35" where login="admin";

select login,pwd from tblUsers;
```
