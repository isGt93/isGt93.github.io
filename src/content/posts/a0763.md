原示例对 XML 内容进行替换；先处理尖括号再处理 \& 可能造成重复转义，使用正则与手动解码也可能损失数据。

```
import os
import re

tags_to_process = ['name', 'published', 'vuln-id', 'severity', 'vuln-type', 'vuln-descript', 'cve-id']


def escape_content_inside_tags(xml_content):
    # 构建一个正则表达式，用于匹配每个指定标签及其内部的内容
    tag_patterns = '|'.join(map(re.escape, tags_to_process))
    pattern = rf'<({tag_patterns})>(.*?)</\1>'

    def escape_text(match):
        # 对匹配到的文本内容进行转义
        return f'<{match.group(1)}>{match.group(2).replace("<", "&lt;").replace(">", "&gt;").replace("&", "&amp;")}</{match.group(1)}>'

    # 使用re.sub函数进行替换
    return re.sub(pattern, escape_text, xml_content, flags=re.DOTALL)
def process_xml_files_in_directory(directory):
    for filename in os.listdir(directory):
        if filename.endswith('.xml'):
            filepath = os.path.join(directory, filename)
            with open(filepath, 'r', encoding='utf-8') as file:
                xml_content = file.read()

            escaped_xml = escape_content_inside_tags(xml_content)

            # 生成新文件名，例如在原文件名前加"escaped_"标记
            new_filename = f"escaped_{filename}"
            new_filepath = os.path.join(directory, new_filename)

            with open(new_filepath, 'w', encoding='utf-8') as new_file:
                new_file.write(escaped_xml)

            print(f'Processed and saved as: {new_filename}')


# 指定要处理的目录
db_directory = 'db'
process_xml_files_in_directory(db_directory)
```

第三步：入库。这里我们编程，将XML文件对应的元数据，存入MongoDB数据库，在存储之前，将特殊符合转换回去，保证和CNNVD下载的XML内容一致

```
import os
import xml.etree.ElementTree as ET
from pymongo import MongoClient, errors

# MongoDB connection details
MONGO_HOST = 'your_ip_address'
MONGO_PORT = 27017
MONGO_USER = 'ailx10'
MONGO_PASS = 'ailx10_passwd'
MONGO_DB = 'db_cnnvd'
MONGO_COLLECTION = 'cnnvd'
AUTH_DB = 'db_cnnvd'
def connect_to_mongodb():
    """Connect to MongoDB and return the collection."""
    try:
        client = MongoClient(f'mongodb://{MONGO_USER}:{MONGO_PASS}@{MONGO_HOST}:{MONGO_PORT}/{MONGO_DB}?authSource={AUTH_DB}')
        db = client[MONGO_DB]
        collection = db[MONGO_COLLECTION]
        print("Connected to MongoDB successfully.")
        return collection
    except errors.PyMongoError as e:
        print(f"Error connecting to MongoDB: {e}")
        raise

def unescape_text(text):
    """Unescape XML/HTML entities in a text."""
    if text:
        text = text.replace("&lt;", "<")
        text = text.replace("&gt;", ">")
        text = text.replace("&amp;", "&")
        text = text.replace("&quot;", "\"")
        text = text.replace("&apos;", "'")
    return text

def get_text_from_element(element, tag):
    """Get text from XML element by tag, unescape entities, and return None if element or tag is not found."""
    sub_element = element.find(tag)
    return unescape_text(sub_element.text) if sub_element is not None else None

def parse_xml(file_path):
    """Parse the XML file and extract entries."""
    tree = ET.parse(file_path)
    root = tree.getroot()
    entries = []
    for entry in root.findall('entry'):
        cve_id = None
        other_id = entry.find('other-id')
        if other_id is not None:
            cve_id_elem = other_id.find('cve-id')
            if cve_id_elem is not None:
                cve_id = unescape_text(cve_id_elem.text)

        entry_data = {
            'name': get_text_from_element(entry, 'name'),
            'published': get_text_from_element(entry, 'published'),
            'vuln-id': get_text_from_element(entry, 'vuln-id'),
            'severity': get_text_from_element(entry, 'severity'),
            'vuln-type': get_text_from_element(entry, 'vuln-type'),
            'vuln-descript': get_text_from_element(entry, 'vuln-descript'),
            'cve-id': cve_id
        }
        entries.append(entry_data)
    return entries


def insert_entries_to_mongodb(entries, collection):
    """Insert entries into MongoDB collection."""
    if entries:
        try:
            collection.insert_many(entries)
        except errors.PyMongoError as e:
            print(f"Error inserting entries to MongoDB: {e}")
            raise

def main():
    db2_directory = 'db2'
    collection = connect_to_mongodb()

    # Iterate over each XML file in the directory
    for file_name in os.listdir(db2_directory):
        if file_name.endswith('.xml'):
            file_path = os.path.join(db2_directory, file_name)
            entries = parse_xml(file_path)
            insert_entries_to_mongodb(entries, collection)
            print(f"Inserted entries from {file_name} into MongoDB.")

if __name__ == '__main__':
    main()
```
