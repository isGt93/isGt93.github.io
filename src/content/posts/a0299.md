### 9、进行coding，暴力破解哈希值

```
# -*- coding: utf-8 -*-
# @Time    : 2022/8/30 9:38 PM
# @Author  : ailx10
# @File    : code.py

import hashlib

f1 = open("iacon_codes","r")
f2 = open("capital","r")

codes = f1.readlines()
capitals = f2.readlines()

for code in codes:
    code = code.strip()
    for capital in capitals:
        capital = capital.strip()
        for i in range(10,100):
            for j in range(10,100):
                str_1 = "/{}/{}/Latitude_{}.{}-Longitude_{}.{}.txt".format(code,capital,i,j,i,j)
                str_2 = hashlib.md5(str_1.encode())
                if str_2.hexdigest() == "daab260727e470e56e77ec22e8f3d413":
                    print(str_1)
```
