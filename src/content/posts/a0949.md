## 2.特征化

每个文件都记录了一长串的系统调用号，每个文件中的系统调用号数量都不一样，并且有大量重复的系统调用号。

我们统计到正常系统调用样本833个，打上0标签，FTP暴力破解系统调用样本162个，打上1标签。

查看`UTD-0001.txt`文件的词集：

```
tr -s "	| " "
" < UTD-0001.txt | sort | uniq -c | sort -n -k 1 -r
```

可以看出：实际次数 系统调用号

```
    539 1
    267 252
      6 6
      2 120
      2 114
      1 63
      1 42
      1 195
```

通过`sklearn`中`CountVectorizer`计算的词集：

```
vectorizer = CountVectorizer(min_df=1)
x=vectorizer.fit_transform(x)
k=vectorizer.get_feature_names()
```

单字符系统调用编号可能被 CountVectorizer 默认 token 模式排除，需核对设置。

应在每个训练折内拟合词表，并按攻击实例分组避免数据泄漏。

```
  (0, 82)	267
  (0, 5)	2
  (0, 47)	1
  (0, 10)	2
  (0, 122)	1
  (0, 128)	1
```

## 5.完整代码

```
import re
import matplotlib.pyplot as plt
import os
from sklearn.feature_extraction.text import CountVectorizer
from sklearn import cross_validation
import os
from sklearn.datasets import load_iris
from sklearn import tree
import pydotplus
import numpy as np

def load_one_flle(filename):
    x=[]
    with open(filename) as f:
        line=f.readline()
        line=line.strip('
')
    return line

def load_adfa_training_files(rootdir):
    x=[]
    y=[]
    list = os.listdir(rootdir)
    for i in range(0, len(list)):
        path = os.path.join(rootdir, list[i])
        if os.path.isfile(path):
            x.append(load_one_flle(path))
            y.append(0)
    
    print len(x)
    return x,y

def dirlist(path, allfile):
    filelist = os.listdir(path)

    for filename in filelist:
        filepath = os.path.join(path, filename)
        if os.path.isdir(filepath):
            dirlist(filepath, allfile)
        else:
            allfile.append(filepath)
    return allfile

def load_adfa_hydra_ftp_files(rootdir):
    x=[]
    y=[]
    allfile=dirlist(rootdir,[])
    for file in allfile:
        if re.match(r"../data/ADFA-LD/Attack_Data_Master/Hydra_FTP_\d+/UAD-Hydra-FTP*",file):
            x.append(load_one_flle(file))
            y.append(1)
    print len(x)
    return x,y



if __name__ == '__main__':
    x1,y1=load_adfa_training_files("../data/ADFA-LD/Training_Data_Master/")
    x2,y2=load_adfa_hydra_ftp_files("../data/ADFA-LD/Attack_Data_Master/")

    x=x1+x2
    y=y1+y2
    vectorizer = CountVectorizer(min_df=1)
    x=vectorizer.fit_transform(x)

    clf = tree.DecisionTreeClassifier()
    score = cross_validation.cross_val_score(clf, x, y, n_jobs=-1, cv=10)
    print score
    print np.mean(score)
    clf = clf.fit(x, y)
    dot_data = tree.export_graphviz(clf, out_file=None)
    graph = pydotplus.graph_from_dot_data(dot_data)
    graph.write_pdf("ftp.pdf")
```

如果同实例或词表信息漏进验证，分数就可能比真实情况乐观。把这些条件补齐后，再看画出的树，才能知道它抓住了哪一种差异。
