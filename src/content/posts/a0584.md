ailx10说：这篇是在2024年11月之前写的，后来又改过。两个演示场景共用一台服务器和同一个端口，每回切换都得登录后台手动停启，做多了实在麻烦。我就写了个 Flask 调度器，让不同网址负责切场景，再交给 Supervisor 守着，异常了能拉起来。后面又从按 PID 停进程，改到按监听端口查找。这篇记录的是演示环境里一个很接地气的小改造：不是新功能有多耀眼，而是把重复操作从手里挪走一点，让下一次演示不必再先钻进后台忙一圈。

---

（1）安装 supervisor

```
sudo apt-get update
sudo apt-get install supervisor
```

（2）配置 supervisor

```
vim /etc/supervisor/conf.d/myshedu.conf
```

编辑内容如下：

```
[program:myshedu]
command=/usr/bin/python3 /root/mymine-test/my_shedu.py
directory=/root/mymine-test
user=root
autostart=true
autorestart=true
stderr_logfile=/var/log/myshedu.err.log
stdout_logfile=/var/log/myshedu.out.log
environment=PATH="/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin",PYTHONUNBUFFERED=1
```

（3）更新 supervisor

```
sudo supervisorctl reread
sudo supervisorctl update
```

（4）管理我的调度服务

```
# 启动
sudo supervisorctl start myshedu

# 停止
sudo supervisorctl stop myshedu

# 重启
sudo supervisorctl restart myshedu

# 查看状态
sudo supervisorctl status
```

```
from flask import Flask, jsonify
import subprocess
import os
import signal
import time

app = Flask(__name__)

PROGRAM1 = "/root/lesuo/main2.py"
PROGRAM2 = "/root/mymine-test/mymine-server.py"

# 存储程序的进程ID
program1_pid = None
program2_pid = None

def start_program(program_path):
    global program1_pid, program2_pid
    if program_path == PROGRAM1:
        # 如果程序2正在运行，则先停止它
        if program2_pid is not None:
            stop_program(PROGRAM2)
        # 启动程序1
        process = subprocess.Popen(["python3", program_path])
        program1_pid = process.pid
    elif program_path == PROGRAM2:
        # 如果程序1正在运行，则先停止它
        if program1_pid is not None:
            stop_program(PROGRAM1)
        # 启动程序2
        process = subprocess.Popen(["python3", program_path])
        program2_pid = process.pid

def stop_program(program_path):
    global program1_pid, program2_pid
    if program_path == PROGRAM1 and program1_pid is not None:
        os.kill(program1_pid, signal.SIGTERM)
        program1_pid = None
    elif program_path == PROGRAM2 and program2_pid is not None:
        os.kill(program2_pid, signal.SIGTERM)
        program2_pid = None

@app.route('/lesuo')
def start_program1():
    start_program(PROGRAM1)
    return jsonify({"status": "lesuo started"})

@app.route('/wakuang')
def start_program2():
    start_program(PROGRAM2)
    return jsonify({"status": "wakuang started"})

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=8910)
```

2更：由于退出程序不优雅，导致绑定的端口已经被其他服务或进程占用，服务端发生了错误：`OSError: [Errno 98] Address already in use` ，解决方案就是等1分钟，再启动另一个进程。

```
def stop_program(program_path):
    global program1_pid, program2_pid
    try:
        if program_path == PROGRAM1 and program1_pid is not None:
            os.kill(program1_pid, signal.SIGKILL)
            os.waitpid(program1_pid, 0)  # 确保进程完全退出
            program1_pid = None
            time.sleep(60)
        elif program_path == PROGRAM2 and program2_pid is not None:
            os.kill(program2_pid, signal.SIGKILL)
            os.waitpid(program2_pid, 0)  # 确保进程完全退出
            program2_pid = None
            time.sleep(60)
    except Exception as e:
        ...
```

完整代码如下：

```
from flask import Flask, request, jsonify
import subprocess
import psutil
import os
import socket
import time
app = Flask(__name__)

# 启动程序的函数
def start_program(program_path, port):
    while is_port_in_use(port):
        print(f"Port {port} is still in use. Waiting to release...")
        time.sleep(1)  # 等待1秒后再检查
    process = subprocess.Popen(['python3', program_path], stdout=subprocess.PIPE, stderr=subprocess.PIPE)
    print(f"{program_path} started on port {port}")

# 停止程序的函数
def stop_program(program_path, port):
    for proc in psutil.process_iter(['pid', 'name']):
        if proc.info['name'] == 'python3' and proc.info['pid'] != os.getpid():
            try:
                p = psutil.Process(proc.info['pid'])
                for conn in p.connections():
                    if conn.status == psutil.CONN_LISTEN and conn.laddr.port == port:
                        os.kill(proc.info['pid'], 9)  # 使用SIGKILL信号强制停止进程
                        print(f"{program_path} stopped")
                        return
            except (psutil.NoSuchProcess, psutil.AccessDenied, psutil.ZombieProcess):
                print("o no.")
    print("No process found running on port {}".format(port))

# 检查端口是否被占用的函数
def is_port_in_use(port):
    with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as s:
        return s.connect_ex(('localhost', port)) == 0


def get_program_on_port(port):
    for proc in psutil.process_iter(['pid', 'name', 'exe','cwd','cmdline']):
        try:
            p = psutil.Process(proc.info['pid'])
            for conn in p.net_connections():

                if conn.status == psutil.CONN_LISTEN and conn.laddr.port == port:
                    # 返回进程的可执行文件路径，这通常是启动的Python脚本
                    print(proc.info)
                    return proc.info['cmdline']
        except (psutil.NoSuchProcess, psutil.AccessDenied, psutil.ZombieProcess):
            pass
    return "No program is currently running on port {}".format(port)

# 路由处理函数
@app.route('/mine', methods=['GET'])
def handle_mine():
    status = request.args.get('status')
    if status == 'start':
        start_program('/root/mymine-test/mymine-server.py', 80)
        return 'MyMine server started', 200
    elif status == 'stop':
        stop_program('/root/mymine-test/mymine-server.py', 80)
        return 'MyMine server stopped', 200
    return 'Invalid status', 400

@app.route('/lesuo', methods=['GET'])
def handle_lesuo():
    status = request.args.get('status')
    if status == 'start':
        start_program('/root/lesuo/main2.py', 80)
        return 'Lesuo server started', 200
    elif status == 'stop':
        stop_program('/root/lesuo/main2.py', 80)
        return 'Lesuo server stopped', 200
    return 'Invalid status', 400

@app.route('/status', methods=['GET'])
def handle_status():
    program_on_port = get_program_on_port(80)
    return jsonify({'program_on_port_80': program_on_port})

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=8910)
```

守护进程解决的是运行状态，不会替接口补认证和并发控制。小工具越方便，越要确认它操作的是正确进程，也别让一个GET请求就能随意改变后台。
