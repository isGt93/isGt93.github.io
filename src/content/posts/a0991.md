1. Python向C\+\+传出反查字典
2. C\+\+解析反查字典，便于记录日志

```

```

PyObject* aisec_waf_init()

{

    Py_Initialize();

    PyRun_SimpleString("import sys");

    PyRun_SimpleString("sys.path.append('./')");

    PyObject* moduleName = PyString_FromString("waf");

    PyObject* pModule = PyImport_Import(moduleName);

    if (!pModule)

    {

        cout << "[waf] Python get module failed." << endl;

        exit(0);

    }

    return pModule;

}



PyObject* aisec_waf_load(PyObject* pModule)

{

    PyObject* waf_load = PyObject_GetAttrString(pModule,"waf_load");

    if (!waf_load || !PyCallable_Check(waf_load))

    {

        cout << "[waf] Can't find funftion (waf_load)" << endl;

        exit(0);

    }

    PyObject* waf_engine = PyObject_CallObject(waf_load,NULL);

    return waf_engine;

}



int aisec_waf_predict(PyObject* pModule,PyObject* waf_engine,const char** pkt_loads)

{

    int isAtk = -1;

    int fields_num = 20;

    PyObject* waf_predict = PyObject_GetAttrString(pModule, "waf_predict");

    if (!waf_predict || !PyCallable_Check(waf_predict))

    {

        cout << "[waf] Can't find funftion (waf_predict)" << endl;

        exit(0);

    }

    PyObject* args = PyTuple_New(2);

    PyObject* arg0 = waf_engine;

    PyObject* arg1 = PyList_New(fields_num);

    for(int i=0; i<fields_num; i++)

    {

        PyList_SetItem(arg1, i, Py_BuildValue("s", pkt_loads[i]));

    }

    

    PyTuple_SetItem(args, 0, arg0);

    PyTuple_SetItem(args, 1, arg1);



    PyObject* waf_predict_info =  PyObject_CallObject(waf_predict,args);

    if (waf_predict_info == NULL)

    {

        cout << "[waf] Can't run funftion (waf_predict)" << endl;

        exit(0);

    }

    // get atk info

    PyObject* arg_atk = PyTuple_New(1);

    PyObject* get_atk_info = PyObject_GetAttrString(pModule, "get_atk_info");

    if(get_atk_info == NULL) cout << "[error]get_atk_info" << endl;

    PyTuple_SetItem(arg_atk, 0, waf_predict_info);

    PyObject* atk_info =  PyObject_CallObject(get_atk_info,arg_atk);

    if(atk_info == NULL) cout << "[error]atk_info" << endl;

    if (atk_info != NULL)

    {

        PyArg_ParseTuple(atk_info,"i",&isAtk);

        cout << "isAtk:" << isAtk << endl;

    }



    // get reverse look info

    PyObject* arg_relook = PyTuple_New(1);

    PyObject* get_reverse_look_info = PyObject_GetAttrString(pModule, "get_reverse_look_info");

    if(get_reverse_look_info == NULL) cout << "[error]get_reverse_look_info" << endl;

    PyTuple_SetItem(arg_relook, 0, waf_predict_info);

    PyObject* reverse_look_info =  PyObject_CallObject(get_reverse_look_info,arg_relook);

    if(reverse_look_info == NULL) cout << "[error]reverse_look_info" << endl;

    aisec_waf_reverse_info(reverse_look_info);

    return isAtk;

}

void aisec_waf_reverse_info(PyObject* reverse_look_info)

{

    if(reverse_look_info == NULL)

        return;

    if(PyDict_Check(reverse_look_info))

    {

        PyObject* pKeys = PyDict_Keys(reverse_look_info);

        for(int i = 0; i < PyList_Size(pKeys); ++i)

        {

            PyObject *pKey = PyList_GetItem(pKeys, i);

            PyObject *pValue = PyDict_GetItem(reverse_look_info, pKey);

            char* s = PyString_AsString(pKey);

            int j = PyInt_AsLong(pValue);

            cout << i << ":" << j << endl;

            cout << "s:" << s << endl;

        }

    }

    return ;

}

class XSS_FEATURE(object):

    def __init__(self):

        return

    def get_len(self, url):

        return len(url)

    def get_url_count(self, url):

        if re.search('(http://)|(https://)', url, re.IGNORECASE) :

            return 1

        else:

            return 0

    def get_evil_char(self, url):

        return len(re.findall("[<>,\'\"/]", url, re.IGNORECASE))

    def get_evil_word(self, url):

        return len(re.findall("(alert)|(script=)(%3c)|(%3e)|(%20)|(onerror)|(onload)|(eval)|(src=)|(prompt)",url,re.IGNORECASE))

def classifier_train(x,y):

    with tf.Graph().as_default():

        x_train, x_test, y_train, y_test=train_test_split( x,y, test_size=0.2,random_state=0)

        x_train = pad_sequences(x_train,maxlen=100,value=0.)

        x_test = pad_sequences(x_test,maxlen=100,value=0.)

        y_train = to_categorical(y_train, nb_classes=4)

        y_test = to_categorical(y_test, nb_classes=4)



        net = tflearn.input_data([None, 100])

        net = tflearn.embedding(net, input_dim=256, output_dim=128)

        net = tflearn.lstm(net, 128, dropout=0.8)

        net = tflearn.fully_connected(net, 4, activation='softmax')

        net = tflearn.regression(net, optimizer='adam', learning_rate=0.1,

                                 loss='categorical_crossentropy')



        model = tflearn.DNN(net, tensorboard_verbose=3)

        model.fit(x_train, y_train,n_epoch=10, validation_set=(x_test, y_test), show_metric=True,

                 batch_size=200,run_id="waf-demo-2018")

        model.save("./model/waf-demo-2018.tfl")
